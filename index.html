import cv2 import numpy as np import math from google.colab import files def step2_embed_to_video(): print("--- 1. ADIM: Az önce indirdiğin 'nike_muhurlu.png' dosyasını seç ---") up_img = files.upload() if not up_img: return noise_img = cv2.imread(list(up_img.keys())[0]) print("\n--- 2. ADIM: Mühürlenecek Videoyu (MP4/MOV) seç ---") up_vid = files.upload() if not up_vid: return vid_path = list(up_vid.keys())[0] # Video Hazırlığı cap = cv2.VideoCapture(vid_path) fps = cap.get(cv2.CAP_PROP_FPS) w, h = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH)), int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT)) total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT)) output_name = 'nike_shop_final.mp4' out = cv2.VideoWriter(output_name, cv2.VideoWriter_fourcc(*'mp4v'), fps, (w, h)) # Esnetmeden tam ortaya yerleştirme ayarları nh, nw, _ = noise_img.shape y_off, x_off = (h - nh) // 2, (w - nw) // 2 # Matematiksel İmza noise_sig = (noise_img.astype(np.float32) / 127.5) - 1.0 STRENGTH = 40 # Okunabilirlik için yüksek şiddet FREQ = 10 # 10Hz Nabız print(f"\n🚀 İşlem başladı: {nw}x{nh} boyutundaki mühür esnetilmeden işleniyor...") frame_count = 0 while cap.isOpened(): ret, frame = cap.read() if not ret: break pulse = (math.sin(2 * math.pi * FREQ * (frame_count / fps)) + 1) / 2 # Sadece merkez bölgeye müdahale et (Esnetme yok!) roi = frame[y_off:y_off+nh, x_off:x_off+nw].astype(np.float32) roi[:,:,0] += noise_sig[:,:,0] * (STRENGTH * pulse) # Mavi roi[:,:,1] -= noise_sig[:,:,1] * (STRENGTH * 0.4 * pulse) # Yeşil denge roi[:,:,2] -= noise_sig[:,:,2] * (STRENGTH * 0.4 * pulse) # Kırmızı denge frame[y_off:y_off+nh, x_off:x_off+nw] = np.clip(roi, 0, 255).astype(np.uint8) out.write(frame) frame_count += 1 if frame_count % 150 == 0: print(f"✅ %{int(frame_count/total_frames*100)} bitti...") cap.release() out.release() print(f"\n🎬 BAŞARILI! Video indiriliyor...") files.download(output_name) step2_embed_to_video()
